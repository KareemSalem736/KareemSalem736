function main a: integer, b: real, c: character returns integer;
    x: integer is 15;
    y: integer is #0A;     -- hex for 10
    r: real is 1.5;
    ch: character is 'B';
    neg: integer is ~1;
    nums: list of integer is (3, 2, 1);
    calc: integer is a + x * y / 2 % 4 ^ 2 + neg;
begin
    if a < 0 then
        when b >= 1.0 & a <= 5, 10 : 20;
    elsif c = 'A' then
        calc;  -- do nothing special, just return calc maybe
    else
        switch a is
            case 1 => fold left + nums endfold;
            case 2 => fold right - nums endfold;
            others => calc;
        endswitch;
    endif;
end;
